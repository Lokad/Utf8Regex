static Matcher BeforeMatcher(Assembly assembly, Case item, Utf8ResourcePolicy policy)
{
    var type = assembly.GetType("Lokad.Utf8Regex.Pcre2.Utf8Pcre2Regex")!; var options = assembly.GetType("Lokad.Utf8Regex.Pcre2.Pcre2CompileOptions")!;
    var settings = assembly.GetType("Lokad.Utf8Regex.Pcre2.Utf8Pcre2CompileSettings")!; var factory = type.GetMethods().Single(m => m.Name == "Create" && m.GetParameters()[0].ParameterType == typeof(string));
    var oldPolicy = Activator.CreateInstance(factory.GetParameters()[3].ParameterType)!;
    foreach (var property in typeof(Utf8ResourcePolicy).GetProperties().Where(p => p.GetMethod is { IsStatic: false }))
        oldPolicy.GetType().GetProperty(property.Name)!.SetValue(oldPolicy, property.GetValue(policy));
    var regex = factory.Invoke(null, [item.Pattern, Enum.ToObject(options, (int)item.Options), Activator.CreateInstance(settings), oldPolicy, Timeout.InfiniteTimeSpan])!;
    var method = type.GetMethods().Single(m => m.Name == (item.Exists ? "IsMatch" : "Count") && !m.IsStatic && m.GetParameters().Length == 4);
    var bridge = new DynamicMethod("BeforePublicOperation", typeof(int), [typeof(object), typeof(byte[]), typeof(CancellationToken)], typeof(Program).Module, true);
    var il = bridge.GetILGenerator(); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); il.Emit(OpCodes.Ldarg_1);
    il.Emit(OpCodes.Call, typeof(ReadOnlySpan<byte>).GetMethod("op_Implicit", [typeof(byte[])])!); il.Emit(OpCodes.Ldc_I4_0); il.Emit(OpCodes.Ldc_I4_0); il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Call, method); il.Emit(OpCodes.Ret);

}
static Func<string, object> BeforeCompiler(Assembly assembly)
{
    var type = assembly.GetType("Lokad.Utf8Regex.Pcre2.Utf8Pcre2Regex")!;
    var method = type.GetMethods().Single(m => m.Name == "Compile" && m.GetParameters().Length == 8 && m.GetParameters()[0].ParameterType == typeof(string));
    var bridge = new DynamicMethod("BeforeCompile", typeof(object), [typeof(string)], typeof(Program).Module, true); var il = bridge.GetILGenerator();
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Ldc_I4_0);

static (double Milliseconds, long Bytes) Loop(int iterations, Func<int> action, int expected)

    for (var i = 0; i < iterations; i++) values[i] = action();

public sealed class Account : IUtf8MemoryAllowance
{
    private long _used, _peak;
    public ulong Used => (ulong)Interlocked.Read(ref _used);
    public ulong Peak => (ulong)Interlocked.Read(ref _peak);
    public bool TryReserve(ulong bytes)
    {
        var used = Interlocked.Add(ref _used, checked((long)bytes));
        long old; do { old = Interlocked.Read(ref _peak); if (used <= old) break; } while (Interlocked.CompareExchange(ref _peak, used, old) != old);
        return true;
    }
    public void Release(ulong bytes) { if (Interlocked.Add(ref _used, -checked((long)bytes)) < 0) throw new Exception("Unfunded release"); }
}
