static (Assembly Core, Assembly Python) Load(string root, string name)
{
    var context = new AssemblyLoadContext(name);
    context.Resolving += (ctx, n) => File.Exists(Path.Combine(root, n.Name + ".dll")) ? ctx.LoadFromAssemblyPath(Path.Combine(root, n.Name + ".dll")) : null;
    return (context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.dll")),
        context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.PythonRe.dll")));
}
static string Hash(Assembly assembly) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(assembly.Location)));
static object Identity(Assembly assembly) => new { Version = assembly.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion, Sha256 = Hash(assembly) };
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes, int Last) Loop(int count, Func<int> action)
{
    var value = 0; var bytes = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) value = action();
    var elapsed = Stopwatch.GetElapsedTime(start);
    return (elapsed.TotalMilliseconds, GC.GetAllocatedBytesForCurrentThread() - bytes, value);
}
