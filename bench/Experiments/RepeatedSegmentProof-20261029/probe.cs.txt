using System.Text;
using System.Text.RegularExpressions;
using Lokad.Utf8Regex;
foreach (var (pattern, text) in new[] {
    (@"(?:[A-Z][a-z]+\s*){2,4}?", "Aa Bb Cc Dd Ee Ff Gg Hh"),
    (@"(?:[a-z][a-z]+\s*){2,4}", "aaaa"),
    (@"(?:[A-Z][a-z]+\s*){2,4}", "Aa BbC Dd Ee") })
foreach (var mode in new[] { RegexOptions.None, RegexOptions.Compiled })
{
    var input = Encoding.UTF8.GetBytes(text); var core = new Utf8Regex(pattern, mode); var oracle = new Regex(pattern, mode);
    var expected = oracle.Match(text); var actual = core.Match(input); var detailed = core.MatchDetailed(input);
    Console.WriteLine($"{pattern}; {mode}; Count={core.Count(input)}/{oracle.Count(text)}; MatchLen={(actual.Success ? actual.LengthInBytes : -1)}/{(expected.Success ? expected.Length : -1)}; DetailedLen={(detailed.Success ? detailed.LengthInBytes : -1)}");
}
