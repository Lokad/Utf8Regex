static Func<object, byte[], CancellationToken, int> Bridge(Type type, bool live) {
    var method = type.GetMethod("Match", live ? [typeof(ReadOnlySpan<byte>), typeof(CancellationToken)] : [typeof(ReadOnlySpan<byte>)])!;
    var bridge = new DynamicMethod("MatchLength", typeof(int), [typeof(object), typeof(byte[]), typeof(CancellationToken)], typeof(Program).Module, true);
    var il = bridge.GetILGenerator(); var value = il.DeclareLocal(method.ReturnType); var success = il.DefineLabel();
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); il.Emit(OpCodes.Ldarg_1); il.Emit(OpCodes.Newobj, typeof(ReadOnlySpan<byte>).GetConstructor([typeof(byte[])])!);
    if (live) il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Call, method); il.Emit(OpCodes.Stloc, value);
    il.Emit(OpCodes.Ldloca, value); il.Emit(OpCodes.Call, method.ReturnType.GetProperty("Success")!.GetMethod!); il.Emit(OpCodes.Brtrue, success);
    il.Emit(OpCodes.Ldc_I4_0); il.Emit(OpCodes.Ret); il.MarkLabel(success); il.Emit(OpCodes.Ldloca, value);
    il.Emit(OpCodes.Call, method.ReturnType.GetProperty("LengthInBytes")!.GetMethod!); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<Func<object, byte[], CancellationToken, int>>();
}
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Measure(int count, Func<int> run, int expected) {
    var values = new int[count]; var bytes = GC.GetAllocatedBytesForCurrentThread(); var stamp = Stopwatch.GetTimestamp();
    for (var index = 0; index < count; index++) values[index] = run();
    var elapsed = Stopwatch.GetElapsedTime(stamp).TotalMilliseconds; bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
    foreach (var value in values) if (value != expected) throw new Exception("Timed result mismatch"); GC.KeepAlive(values); return (elapsed, bytes);
}
