static (Assembly Core, Assembly Adapter) Load(string root, string name)
{
    var context = new AssemblyLoadContext(name);
    context.Resolving += (ctx, n) => File.Exists(Path.Combine(root, n.Name + ".dll")) ? ctx.LoadFromAssemblyPath(Path.Combine(root, n.Name + ".dll")) : null;
    return (context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.dll")), context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.Pcre2.dll")));
}
static string Hash(Assembly a) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(a.Location)));
static object Identity(Assembly a) => new { Version = a.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion, Sha256 = Hash(a) };
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes, int Last) Loop(int count, Func<int> action)
{

    for (var i = 0; i < count; i++) values[i] = action();

    bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
    var value = values[0]; foreach (var current in values) if (current != value) throw new Exception("Result batch mismatch");
    GC.KeepAlive(values); return (duration, bytes, value);
}
