static Func<object, byte[], int> CountBridge(Type type) {
    var bridge = new DynamicMethod("PublicCount", typeof(int), [typeof(object), typeof(byte[])], typeof(Program).Module, true); var il = bridge.GetILGenerator();
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); ReadSpan(il, 1); il.Emit(OpCodes.Call, type.GetMethod("Count", [typeof(ReadOnlySpan<byte>)])!); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<Func<object, byte[], int>>();
}
static Func<object, byte[], object, CancellationToken, object> ReplaceBridge(Type type, bool native, bool text) {

[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Loop(int count, Func<Output> action, Action<Output> check) {
    var output = new Output[count]; var bytes = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp(); for (var i = 0; i < count; i++) output[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(start).TotalMilliseconds; bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
    foreach (var value in output) check(value); GC.KeepAlive(output); return (elapsed, bytes);
}
