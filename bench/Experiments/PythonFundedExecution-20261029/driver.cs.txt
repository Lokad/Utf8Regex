static Replace ReplaceBridge(Type type,bool core)
{

static Collector CursorBridge(Type type)
{

    var groupsGetter=currentType.GetProperty("Groups")!.GetMethod!;var groupType=groupsGetter.ReturnType.GetElementType()!;
    var dm=new DynamicMethod("PublicCursor",typeof(string[]),[typeof(object),typeof(byte[]),typeof(CancellationToken)],typeof(Program).Module,true);var il=dm.GetILGenerator();

    il.Emit(OpCodes.Newobj,typeof(List<string>).GetConstructor(Type.EmptyTypes)!);il.Emit(OpCodes.Stloc,values);
    il.Emit(OpCodes.Ldarg_0);il.Emit(OpCodes.Castclass,type);il.Emit(OpCodes.Ldarg_1);il.Emit(OpCodes.Newobj,typeof(ReadOnlyMemory<byte>).GetConstructor([typeof(byte[])])!);
    il.Emit(OpCodes.Ldc_I4_0);il.Emit(OpCodes.Ldarg_2);il.Emit(OpCodes.Callvirt,factory);il.Emit(OpCodes.Stloc,cursor);
    il.BeginExceptionBlock();var move=il.DefineLabel();var next=il.DefineLabel();il.Emit(OpCodes.Br,move);il.MarkLabel(next);

    il.Emit(OpCodes.Ldloc,values);il.Emit(OpCodes.Ldloca,current);il.Emit(OpCodes.Call,groupsGetter);il.Emit(OpCodes.Ldc_I4_0);il.Emit(OpCodes.Ldelema,groupType);
    il.Emit(OpCodes.Call,groupType.GetProperty("ValueText")!.GetMethod!);il.Emit(OpCodes.Callvirt,typeof(List<string>).GetMethod("Add")!);
    il.MarkLabel(move);il.Emit(OpCodes.Ldloc,cursor);il.Emit(OpCodes.Callvirt,cursorType.GetMethod("MoveNext",Type.EmptyTypes)!);il.Emit(OpCodes.Brtrue,next);
    il.BeginFinallyBlock();il.Emit(OpCodes.Ldloc,cursor);il.Emit(OpCodes.Callvirt,cursorType.GetMethod("Dispose")!);il.EndExceptionBlock();

static object MakeAllowance(Type contract,Host host)

    for(var i=0;i<count;i++)retained[i]=action();

    long checksum=0;foreach(var result in retained){if(!expected.Any(e=>Equal(e,result)))throw new Exception("Timed result mismatch");checksum=unchecked(checksum+result.Count+(result.Text?.GetHashCode()??0));if(result.Values is not null)foreach(var text in result.Values)checksum=unchecked(checksum+text.GetHashCode());}
