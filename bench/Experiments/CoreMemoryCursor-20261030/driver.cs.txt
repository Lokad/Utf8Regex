static long Detailed(Utf8Regex regex, byte[] input, int[] numbers, CancellationToken token)
{
    var floor = 0; long digest = 17;
    while (floor <= input.Length)
    {
        var match = regex.MatchDetailedFromUtf16Offset(input, floor, token); if (!match.Success) break;
        foreach (var number in numbers)
        {
            var group = match.GetGroup(number); digest = Mix(Mix(digest, number), group.CaptureCount);
            for (var i = 0; i < group.CaptureCount; i++) { var capture = group.GetCapture(i); digest = Mix(Mix(digest, capture.IndexInBytes), capture.IndexInBytes + capture.LengthInBytes); }
        }
        if (match.LengthInUtf16 == 0) throw new Exception("This bounded pilot requires nonempty patterns");
        floor = match.IndexInUtf16 + match.LengthInUtf16;
    }
    return digest;
}
static long Ranges(Utf8RegexMatchCursor cursor, Account? result = null)
{
    using (cursor)
    {
        long digest = 17;
        if (result is null) while (cursor.MoveNext(out var match)) digest = Digest(digest, match);
        else while (cursor.MoveNextRetained(out var receipt, resultMemoryAllowance: result)) { using (receipt) digest = Digest(digest, receipt.Match); }
        return digest;
    }
}
static long Digest(long digest, Utf8RangeMatch match)
{
    foreach (var number in match.GroupNumbers)
    {
        var captures = match.GetCaptures(number); digest = Mix(Mix(digest, number), captures.Length);
        foreach (var capture in captures) digest = Mix(Mix(digest, capture.StartOffsetInBytes), capture.EndOffsetInBytes);
    }
    return digest;
}
static long Mix(long digest, int value) => unchecked((digest * 31) ^ value);
static (double Milliseconds, long Bytes) Loop(int iterations, Func<long> action, long expected)
{

    for (var i = 0; i < iterations; i++) values[i] = action();

    long checksum = 0; foreach (var value in values) { if (value != expected) throw new Exception("Timed result mismatch"); checksum ^= value; }
    GC.KeepAlive(checksum); return (elapsed, bytes);
}

public sealed class Account : IUtf8MemoryAllowance
{
    private long _used, _peak;
    public ulong Used => (ulong)Interlocked.Read(ref _used);
    public ulong Peak => (ulong)Interlocked.Read(ref _peak);
    public bool TryReserve(ulong bytes)
    {
        var used = Interlocked.Add(ref _used, checked((long)bytes));
        long old; do { old = Interlocked.Read(ref _peak); if (used <= old) break; } while (Interlocked.CompareExchange(ref _peak, used, old) != old);
        return true;
    }
    public void Release(ulong bytes) { if (Interlocked.Add(ref _used, -checked((long)bytes)) < 0) throw new Exception("Unfunded release"); }
}
