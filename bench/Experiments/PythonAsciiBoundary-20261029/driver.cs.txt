static object Run<T>(object meta, Func<T>[] actions, Func<T, ulong> digest)
{
    var expected = digest(actions[2]());
    var counts = new int[4]; var warmCalls = new int[4]; var warmMs = new double[4];
    for (var lane = 0; lane < 4; lane++)
    {
        if (digest(actions[lane]()) != expected) throw new Exception("Preflight mismatch: " + JsonSerializer.Serialize(meta));
        var warm = Stopwatch.StartNew();
        do { Measure(1, actions[lane], digest, expected); warmCalls[lane]++; } while (warm.ElapsedMilliseconds < 1000);
        warmMs[lane] = warm.Elapsed.TotalMilliseconds;
        var n = 1; var p = Measure(n, actions[lane], digest, expected);
        while (p.Milliseconds < 20 && n < 65536 && p.Bytes < 33554432) { n *= 2; p = Measure(n, actions[lane], digest, expected); }
        var cap = p.Bytes > 0 ? Math.Min(65536, (int)Math.Max(1, 67108864L * n / p.Bytes)) : 65536;
        counts[lane] = (int)Math.Clamp(Math.Ceiling(n * 50 / Math.Max(p.Milliseconds, 0.000001)), 1, cap);
    }
    var times = Enumerable.Range(0, 4).Select(_ => new double[3]).ToArray();
    var bytes = Enumerable.Range(0, 4).Select(_ => new double[3]).ToArray();
    for (var sample = 0; sample < 3; sample++)
    for (var position = 0; position < 4; position++)
    {
        var lane = sample % 2 == 0 ? position : 3 - position;
        GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
        var result = Measure(counts[lane], actions[lane], digest, expected);
        times[lane][sample] = result.Milliseconds * 1000 / counts[lane];
        bytes[lane][sample] = (double)result.Bytes / counts[lane];
    }
    Console.Error.WriteLine(JsonSerializer.Serialize(meta) + " : " + string.Join(" / ", times.Select(x => x.Order().ElementAt(1))) + " us/op");
    return new { Workload = meta, ExpectedDigest = expected, Iterations = counts, WarmupCalls = warmCalls,
        WarmupMilliseconds = warmMs, Microseconds = times, AllocatedBytes = bytes };
}
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Measure<T>(int count, Func<T> action, Func<T, ulong> digest, ulong expected)
{
    var retained = new T[count];
    var bytes = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) retained[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(start); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
    foreach (var value in retained) if (digest(value) != expected) throw new Exception("Timed digest mismatch");
    GC.KeepAlive(retained);
    return (elapsed.TotalMilliseconds, bytes);
}
static ulong Digest(string?[] values)
{
    ulong hash = 14695981039346656037UL;
    unchecked
    {
        foreach (var value in values)
        {
            hash = (hash ^ (uint)(value?.Length ?? -1)) * 1099511628211UL;
            if (value is not null) foreach (var c in value) hash = (hash ^ c) * 1099511628211UL;
        }
    }
    return hash;
}
static (Assembly Core, Assembly Python) Load(string root, string name)
{
    var context = new AssemblyLoadContext(name);
    context.Resolving += (ctx, n) => File.Exists(Path.Combine(root, n.Name + ".dll")) ? ctx.LoadFromAssemblyPath(Path.Combine(root, n.Name + ".dll")) : null;
    return (context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.dll")), context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.PythonRe.dll")));
}
static string Version(Assembly assembly) => assembly.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion;
static string Hash(Assembly assembly) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(assembly.Location)));
static object Identity((Assembly Core, Assembly Python) pair) => new { Core = new { Version = Version(pair.Core), Sha256 = Hash(pair.Core) }, Python = new { Version = Version(pair.Python), Sha256 = Hash(pair.Python) } };
