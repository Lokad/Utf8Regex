static GroupRange[][] Oracle(Regex regex,string text)
{
    var values=new List<GroupRange[]>();
    foreach(Match match in regex.Matches(text))
    {
        var groups=new GroupRange[match.Groups.Count];
        for(var i=0;i<groups.Length;i++)
        {
            var g=match.Groups[i];groups[i]=g.Success?new(i,true,Encoding.UTF8.GetByteCount(text.AsSpan(0,g.Index)),Encoding.UTF8.GetByteCount(text.AsSpan(0,g.Index+g.Length))):new(i,false,0,0);
        }
        values.Add(groups);
    }
    return values.ToArray();
}
