static object Measure(string[] lanes,Func<int,Sample>[] runs,Action balances)
{
    var iterations=new int[runs.Length];var warmCalls=new int[runs.Length];
    for(var lane=0;lane<runs.Length;lane++)
    {
        var warm=Stopwatch.StartNew();do{_=runs[lane](1);warmCalls[lane]++;}while(warm.Elapsed.TotalMilliseconds<600);
        iterations[lane]=1;while(runs[lane](iterations[lane]).Milliseconds<80)iterations[lane]=checked(iterations[lane]*2);
    }
    var times=runs.Select(_=>new double[3]).ToArray();var bytes=runs.Select(_=>new double[3]).ToArray();var durations=runs.Select(_=>new double[3]).ToArray();
    for(var sample=0;sample<3;sample++)for(var position=0;position<runs.Length;position++)
    {
        var lane=sample%2==0?position:runs.Length-1-position;GC.Collect();GC.WaitForPendingFinalizers();GC.Collect();var value=runs[lane](iterations[lane]);
        times[lane][sample]=value.Milliseconds*1000/iterations[lane];bytes[lane][sample]=(double)value.Bytes/iterations[lane];durations[lane][sample]=value.Milliseconds;balances();
    }
    Console.Error.WriteLine(string.Join(" / ",times.Select(a=>a.Order().ElementAt(1))));
    return new {Lanes=lanes,Iterations=iterations,WarmupCalls=warmCalls,Microseconds=times,AllocatedBytes=bytes,ElapsedMilliseconds=durations};
}
static Witness[] Collect(Utf8Pcre2Regex regex,byte[] input,CancellationToken token)
{
    var buffer=new Witness[4096];var written=Detailed(regex,input,token,buffer);return buffer[..written];
}
static int Detailed(Utf8Pcre2Regex regex,byte[] input,CancellationToken token,Span<Witness> destination)
{
    var floor=0;var written=0;
    while(floor<=input.Length)
    {
        var match=regex.MatchDetailed(input,floor,default,token);if(!match.Success)break;
        for(var slot=0;slot<match.CaptureSlotCount;slot++)
        {var group=match.GetGroup(slot);destination[written++]=new(slot,group.StartOffsetInBytes,group.EndOffsetInBytes,group.Success,group.IsUtf8SliceWellFormed);}
        if(match.Value.StartOffsetInBytes==match.Value.EndOffsetInBytes)throw new Exception("Pilot requires nonempty matches");
        floor=match.Value.EndOffsetInBytes;
    }
    return written;
}
static int Ranges(Utf8Pcre2MatchCursor cursor,CancellationToken token,Span<Witness> destination,Account? result=null)
{
    using(cursor)
    {
        var written=0;
        if(result is null)while(cursor.MoveNext(out var match,token))written=Write(match,destination,written);
        else while(cursor.MoveNextRetained(out var receipt,token,result)){using(receipt)written=Write(receipt.Match,destination,written);}
        return written;
    }
}
static int Write(Utf8Pcre2RangeMatch match,Span<Witness> destination,int written)
{
    foreach(var group in match.Groups)destination[written++]=new(group.Number,group.StartOffsetInBytes,group.EndOffsetInBytes,group.Success,group.IsUtf8SliceWellFormed);
    return written;
}
static Sample CaptureLoop(int count,CaptureRun run,Witness[] expected)
{
    var values=new Witness[checked(count*expected.Length)];var written=new int[count];var allocation=GC.GetAllocatedBytesForCurrentThread();var start=Stopwatch.GetTimestamp();
    for(var index=0;index<count;index++)written[index]=run(values.AsSpan(index*expected.Length,expected.Length));
    var elapsed=Stopwatch.GetElapsedTime(start).TotalMilliseconds;var bytes=GC.GetAllocatedBytesForCurrentThread()-allocation;
    long checksum=0;for(var index=0;index<count;index++)
    {
        if(written[index]!=expected.Length)throw new Exception("Capture count mismatch");
        for(var slot=0;slot<expected.Length;slot++){var value=values[index*expected.Length+slot];if(value!=expected[slot])throw new Exception("Capture range mismatch");checksum+=value.Start+value.End;}
    }
    GC.KeepAlive(checksum);return new(elapsed,bytes);
}
static Sample CountLoop(int count,Func<int> run,int expected)
{
    var values=new int[count];var allocation=GC.GetAllocatedBytesForCurrentThread();var start=Stopwatch.GetTimestamp();
    for(var index=0;index<count;index++)values[index]=run();
    var elapsed=Stopwatch.GetElapsedTime(start).TotalMilliseconds;var bytes=GC.GetAllocatedBytesForCurrentThread()-allocation;
    long checksum=0;foreach(var value in values){if(value!=expected)throw new Exception("Count mismatch");checksum+=value;}GC.KeepAlive(checksum);return new(elapsed,bytes);
}
static Sample ReplaceLoop(int count,Func<byte[]> run,byte[] expected)
{
    var values=new byte[count][];var allocation=GC.GetAllocatedBytesForCurrentThread();var start=Stopwatch.GetTimestamp();
    for(var index=0;index<count;index++)values[index]=run();
    var elapsed=Stopwatch.GetElapsedTime(start).TotalMilliseconds;var bytes=GC.GetAllocatedBytesForCurrentThread()-allocation;
    long checksum=0;foreach(var value in values){if(!value.AsSpan().SequenceEqual(expected))throw new Exception("Replacement mismatch");foreach(var part in value)checksum+=part;}
    GC.KeepAlive(checksum);return new(elapsed,bytes);
}
