using System.Reflection;
using System.Text;
using System.Text.Json;
using System.Text.RegularExpressions;
using Lokad.Utf8Regex;

var cases=new[]{("^[0-9]{4}-[0-9]{2}-[0-9]{2}$","2026-10-23"),("^ab[0-9]{2}$","ab12"),("(?:cat|dog)[0-9]+","cat12")};
var rows=new List<object>();
foreach(var item in cases)
foreach(var finite in new[]{false,true})
foreach(var compiled in new[]{false,true})
{
    var policy=finite ? new Utf8ResourcePolicy{MaxWorkSteps=1_000_000,MaxDepth=1024,MaxWorkingBytes=16<<20,MaxResultBytes=16<<20}:default;
    var options=RegexOptions.CultureInvariant|(compiled?RegexOptions.Compiled:0);
    Utf8Regex Create()=>new(item.Item1,options,Timeout.InfiniteTimeSpan,policy);
    _=Create();
    var allocated=GC.GetAllocatedBytesForCurrentThread();
    var regex=Create();
    allocated=GC.GetAllocatedBytesForCurrentThread()-allocated;
    var program=typeof(Utf8Regex).GetField("_program",BindingFlags.Instance|BindingFlags.NonPublic)!.GetValue(regex)!;
    var runtime=program.GetType().GetProperty("CompiledEngineRuntime")!.GetValue(program)!;
    var emitted=new List<object>();
    for(var type=runtime.GetType();type!=null;type=type.BaseType)
        foreach(var field in type.GetFields(BindingFlags.Instance|BindingFlags.NonPublic|BindingFlags.DeclaredOnly))
            if(field.Name.Contains("emitted",StringComparison.OrdinalIgnoreCase) && field.GetValue(runtime) is {} value)
                emitted.Add(new{Field=field.Name,Type=value.GetType().Name});
    var input=Encoding.UTF8.GetBytes(item.Item2);
    if(!regex.IsMatch(input)||regex.Count(input)!=1||!regex.Match(input).Success) throw new Exception("Public mismatch");
    if(regex.MatchDetailed(input).GetValueString()!=item.Item2||regex.ReplaceToString(input,"X")!="X") throw new Exception("Result mismatch");
    var cursor=regex.EnumerateMatches(input);
    if(!cursor.MoveNext()||cursor.MoveNext()) throw new Exception("Enumeration mismatch");
    var destination=new byte[32];
    if(regex.TryReplace(input,"X",destination,out var written)!=System.Buffers.OperationStatus.Done||written!=1||destination[0]!='X') throw new Exception("Destination mismatch");
    rows.Add(new{Pattern=item.Item1,Finite=finite,Compiled=compiled,Runtime=runtime.GetType().Name,ConstructorAllocatedBytes=allocated,EmittedRoots=emitted,PublicOperationsAgree=true});
}
Console.WriteLine(JsonSerializer.Serialize(new{Scope="Current construction allocation and retained emitted-root witness; no throughput or changed implementation",Producer=typeof(Utf8Regex).Assembly.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion,Rows=rows},new JsonSerializerOptions{WriteIndented=true}));
