var reset = new Utf8Pcre2Regex(resetPattern);

var lanes = new[] {
    new Lane("construction/literal", () => Construct("needle"), 1, 64),
    new Lane("construction/named-branch-reset", () => Construct(resetPattern), 2, 64),
    new Lane("construction/sixteen-captures", () => Construct(capturesPattern), 1, 64),
    new Lane("public/Analyze-branch-reset", () => reset.Analyze().MinRequiredLengthInBytes, 2, 100_000),
    new Lane("public/Count-branch-reset", () => reset.Count(bytes), 256, 64),
    new Lane("control/unchanged-Core-Count", () => literal.Count(bytes), 128, 1024),
    new Lane("control/predecoded-Regex-Count", () => oracle.Count(text), 128, 1024),

}

static int Construct(string pattern)
{
    var regex = new Utf8Pcre2Regex(pattern);
    var count = regex.NameEntryCount;
    GC.KeepAlive(regex);
    return count + 1;
}
static Sample Run(Lane lane, int iterations)
{
    var allocated = GC.GetAllocatedBytesForCurrentThread();
    var start = Stopwatch.GetTimestamp();
    long checksum = 0;
    for (var iteration = 0; iteration < iterations; iteration++) checksum += lane.Body();
    var elapsed = Stopwatch.GetElapsedTime(start).TotalMilliseconds;
    allocated = GC.GetAllocatedBytesForCurrentThread() - allocated;
    if (checksum != (long)lane.Expected * iterations) throw new Exception("Result mismatch: " + lane.Id);
    return new Sample(elapsed, elapsed * 1_000_000 / iterations, (double)allocated / iterations, checksum);
}
sealed class Lane(string id, Func<int> body, int expected, int minimum)
{
    internal string Id { get; } = id;
    internal Func<int> Body { get; } = body;
    internal int Expected { get; } = expected;
    internal int Minimum { get; } = minimum;
    internal int Iterations { get; set; }
    internal List<Sample> Samples { get; } = [];
}
sealed record Sample(double Milliseconds, double NanosecondsPerOperation, double AllocatedBytesPerOperation, long Checksum);
