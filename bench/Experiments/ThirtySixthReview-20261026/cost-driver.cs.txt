using System.Buffers;
using System.Reflection;
using System.Text;
using System.Text.RegularExpressions;
using Lokad.Utf8Regex;
using Lokad.Utf8Regex.Pcre2;

public static class Lanes
{
    private static object? _retained;
    private static Utf8Pcre2Regex[]? _constructed;
    private static readonly Utf8ResourcePolicy Policy = new() { MaxWorkSteps=50_000_000, MaxDepth=1024,
        MaxWorkingBytes=64*1024*1024, MaxResultBytes=128*1024*1024 };

    public static string Producer() => typeof(Utf8Pcre2Regex).Assembly.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion;
    private static Utf8Pcre2Regex Regex(string pattern, bool finite) => finite
        ? Utf8Pcre2Regex.Create(pattern,default,default,Policy,Timeout.InfiniteTimeSpan) : new(pattern);

    public static (string,Func<long>,long)[] Create()
    {
        var lanes = new List<(string,Func<long>,long)>();
        foreach(var finite in new[]{false,true})
        {
            var prefix = finite ? "finite/" : "unlimited/";
            AddBytes(lanes,prefix+"short-bytes",Regex("a",finite),"aba","$0","aba");
            var dense = new string('a',1024);
            AddBytes(lanes,prefix+"dense-bytes",Regex("a",finite),dense,"$0",dense);
            var unicode = string.Concat(Enumerable.Repeat("é😀 ",128));
            AddText(lanes,prefix+"unicode-text",Regex("é",finite),unicode,"[$0]",unicode.Replace("é","[é]",StringComparison.Ordinal));
        }
        var sparse = new string('x',16384)+"abc"+new string('x',64);
        AddBytes(lanes,"unlimited/sparse-bytes",Regex("abc",false),sparse,"$0_",sparse.Replace("abc","abc_",StringComparison.Ordinal));
        var flagsInput = new string('a',1024);
        AddBytes(lanes,"finite/runtime-flags-streaming",Regex("a",true),flagsInput,"$0",flagsInput,
            Pcre2SubstitutionOptions.UnsetEmpty|Pcre2SubstitutionOptions.UnknownUnset);
        var backtracking = string.Concat(Enumerable.Repeat("ab ",128));
        AddBytes(lanes,"unlimited/backtracking-ledger",Regex("(?<x>a)b",false),backtracking,"$0_",backtracking.Replace("ab","ab_",StringComparison.Ordinal));
        var destinationRegex = Regex("a",true);
        var input = Encoding.UTF8.GetBytes(flagsInput);
        var destination = new byte[input.Length];
        if(destinationRegex.TryReplace(input,"$0"u8,destination,out var written)!=OperationStatus.Done || written!=input.Length || !destination.AsSpan().SequenceEqual(input))
            throw new Exception("Destination preflight mismatch");
        lanes.Add(("finite/destination-ledger",()=>Destination(destinationRegex,input,destination),ByteChecksum(input)));
        var control = new Utf8Regex("abc",RegexOptions.CultureInvariant|RegexOptions.Compiled);
        lanes.Add(("Unchanged-control/Core-compiled-literal-Count",()=>control.Count("abc xabc"u8),2));
        return lanes.ToArray();
    }

    private static void AddBytes(List<(string,Func<long>,long)> lanes,string name,Utf8Pcre2Regex regex,
        string text,string replacement,string expected,Pcre2SubstitutionOptions options=default)
    {
        var input=Encoding.UTF8.GetBytes(text); var expectedBytes=Encoding.UTF8.GetBytes(expected);
        if(!regex.Replace(input,replacement,0,options).AsSpan().SequenceEqual(expectedBytes)) throw new Exception("Full byte preflight: "+name);
        lanes.Add((name,()=>Bytes(regex,input,replacement,options),ByteChecksum(expectedBytes)));
    }
    private static long Bytes(Utf8Pcre2Regex regex,byte[] input,string replacement,Pcre2SubstitutionOptions options)
    { var result=regex.Replace(input,replacement,0,options); _retained=result; return ByteChecksum(result); }
    private static void AddText(List<(string,Func<long>,long)> lanes,string name,Utf8Pcre2Regex regex,string text,string replacement,string expected)
    {
        var input=Encoding.UTF8.GetBytes(text);
        if(regex.ReplaceToString(input,replacement)!=expected) throw new Exception("Full text preflight: "+name);
        lanes.Add((name,()=>Text(regex,input,replacement),TextChecksum(expected)));
    }
    private static long Text(Utf8Pcre2Regex regex,byte[] input,string replacement)
    { var result=regex.ReplaceToString(input,replacement); _retained=result; return TextChecksum(result); }
    private static long Destination(Utf8Pcre2Regex regex,byte[] input,byte[] destination)
    {
        if(regex.TryReplace(input,"$0"u8,destination,out var written)!=OperationStatus.Done) throw new Exception("Destination timed failure");
        return ByteChecksum(destination.AsSpan(0,written));
    }
    private static long ByteChecksum(ReadOnlySpan<byte> bytes) => bytes.Length+(bytes.IsEmpty?0:bytes[0]+bytes[^1]);
    private static long TextChecksum(string text) => text.Length+(text.Length==0?0:text[0]+text[^1]);

    public static object Construction()
    {
        for(var i=0;i<2;i++) _retained=new Utf8Pcre2Regex("a");
        var instances=new Utf8Pcre2Regex[8];
        var before=GC.GetAllocatedBytesForCurrentThread();
        for(var i=0;i<instances.Length;i++) instances[i]=new Utf8Pcre2Regex("a");
        var bytes=GC.GetAllocatedBytesForCurrentThread()-before;
        _constructed=instances;
        return new { BytesPerConstructor=bytes/instances.Length, Instances=instances.Length };
    }
}
