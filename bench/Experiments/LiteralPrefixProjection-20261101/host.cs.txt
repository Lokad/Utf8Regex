[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Measure(int count, Func<Value> action, Value expected) {
    var values = new Value[count]; var allocated = GC.GetAllocatedBytesForCurrentThread(); var stamp = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) values[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(stamp).TotalMilliseconds; allocated = GC.GetAllocatedBytesForCurrentThread() - allocated;
    foreach (var value in values) if (value != expected) throw new Exception("Timed output mismatch");
    GC.KeepAlive(values); return (elapsed, allocated);
}
