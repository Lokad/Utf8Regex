static Pair Load(string root, string name)
{
    var context = new AssemblyLoadContext(name);
    var core = context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.dll"));
    var pcre = context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.Pcre2.dll"));
    return new(core, pcre);
}
static object Policy(Assembly assembly, bool finite)
{
    var type = assembly.GetType("Lokad.Utf8Regex.Utf8ResourcePolicy")!; var value = Activator.CreateInstance(type)!;
    if (finite) { type.GetProperty("MaxWorkSteps")!.SetValue(value, 16_000_000UL); type.GetProperty("MaxDepth")!.SetValue(value, 4096U);
        type.GetProperty("MaxWorkingBytes")!.SetValue(value, 4UL << 20); type.GetProperty("MaxResultBytes")!.SetValue(value, 4UL << 20); }
    return value;
}
static Compiler CompileBridge(Type type, Assembly core, bool native, bool funded, string options)
{
    var isPcre = type.Name == "Utf8Pcre2Regex";
    var method = type.GetMethods().Single(m => m.Name == "Compile" && m.GetParameters().Length == (isPcre ? (funded ? 9 : 8) : (funded ? 8 : 7)) && m.GetParameters()[0].ParameterType == (native ? typeof(ReadOnlySpan<byte>) : typeof(string)));
    var bridge = new DynamicMethod("CompilePublic", typeof(object), [typeof(object), typeof(object), typeof(object), typeof(object)], typeof(Program).Module, true);
    var il = bridge.GetILGenerator(); var token = il.DeclareLocal(typeof(CancellationToken));
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, native ? typeof(byte[]) : typeof(string));
    if (native) il.Emit(OpCodes.Call, typeof(ReadOnlySpan<byte>).GetMethod("op_Implicit", [typeof(byte[])])!);
    il.Emit(OpCodes.Ldc_I4, Convert.ToInt32(Enum.Parse(method.GetParameters()[1].ParameterType, options)));
    if (isPcre) il.Emit(OpCodes.Ldloc, il.DeclareLocal(method.GetParameters()[2].ParameterType));
    il.Emit(OpCodes.Ldsfld, typeof(Timeout).GetField("InfiniteTimeSpan")!);
    il.Emit(OpCodes.Ldarg_1); il.Emit(OpCodes.Unbox_Any, core.GetType("Lokad.Utf8Regex.Utf8ResourcePolicy")!);
    il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Unbox_Any, core.GetType("Lokad.Utf8Regex.Utf8ConstructionPolicy")!);
    il.Emit(OpCodes.Ldsfld, typeof(Timeout).GetField("InfiniteTimeSpan")!); il.Emit(OpCodes.Ldloc, token);
    if (funded) { il.Emit(OpCodes.Ldarg_3); il.Emit(OpCodes.Castclass, core.GetType("Lokad.Utf8Regex.IUtf8MemoryAllowance")!); }
    il.Emit(OpCodes.Call, method); il.Emit(OpCodes.Ret); return bridge.CreateDelegate<Compiler>();
}

static SpanCount CountBridge(object regex)
{
    var type = regex.GetType(); var method = type.GetMethod("Count", [typeof(ReadOnlySpan<byte>)]);
    if (method is not null) return method.CreateDelegate<SpanCount>(regex);
    method = type.GetMethods().Single(m => m.Name == "Count");
    var bridge = new DynamicMethod("PublicFundedCount", typeof(int), [typeof(object), typeof(ReadOnlySpan<byte>)], typeof(Program).Module, true);
    var il = bridge.GetILGenerator(); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); il.Emit(OpCodes.Ldarg_1);
    foreach (var parameter in method.GetParameters().Skip(1))
    {
        if (parameter.ParameterType == typeof(CancellationToken)) il.Emit(OpCodes.Ldloc, il.DeclareLocal(typeof(CancellationToken)));
        else if (parameter.ParameterType.IsValueType) il.Emit(OpCodes.Ldc_I4_0);
        else il.Emit(OpCodes.Ldnull);
    }
    il.Emit(OpCodes.Call, method); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<SpanCount>(regex);
}
static object MakeAllowance(Type contract, Host host)
{

    var type = assembly.DefineDynamicModule("Host").DefineType("Provider", TypeAttributes.Public | TypeAttributes.Sealed, typeof(object), [contract]);
    var field = type.DefineField("host", typeof(Host), FieldAttributes.Private); var ctor = type.DefineConstructor(MethodAttributes.Public, CallingConventions.Standard, [typeof(Host)]);
    var il = ctor.GetILGenerator(); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Call, typeof(object).GetConstructor(Type.EmptyTypes)!); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Ldarg_1); il.Emit(OpCodes.Stfld, field); il.Emit(OpCodes.Ret);
    foreach (var declaration in contract.GetMethods()) { var method = type.DefineMethod(declaration.Name, MethodAttributes.Public | MethodAttributes.Virtual | MethodAttributes.Final | MethodAttributes.NewSlot, declaration.ReturnType, [typeof(ulong)]);
        il = method.GetILGenerator(); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Ldfld, field); il.Emit(OpCodes.Ldarg_1); il.Emit(OpCodes.Callvirt, typeof(Host).GetMethod(declaration.Name)!); il.Emit(OpCodes.Ret); type.DefineMethodOverride(method, declaration); }
    return Activator.CreateInstance(type.CreateType()!, host)!;
}

[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Loop(int count, Func<int> action, int expected)
{
    var output = new int[count]; var allocated = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) output[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(start).TotalMilliseconds; var bytes = GC.GetAllocatedBytesForCurrentThread() - allocated;
    long checksum = 0; foreach (var value in output) { if (value != expected) throw new Exception("Timed mismatch"); checksum = unchecked(checksum + value); }
    GC.KeepAlive(output); GC.KeepAlive(checksum); return (elapsed, bytes);
}

{

    public bool TryReserve(ulong bytes) { if (bytes > long.MaxValue) return false; var delta = (long)bytes; while (true) { var current = Volatile.Read(ref _used); if (delta > long.MaxValue - current) return false; if (Interlocked.CompareExchange(ref _used, current + delta, current) == current) return true; } }
