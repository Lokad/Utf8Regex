using System.Diagnostics;
using System.Reflection;
using System.Reflection.Emit;
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using System.Runtime.Loader;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Text.RegularExpressions;

var hostRoot = Path.GetFullPath(args[0]);
var candidatePath = Path.GetFullPath(args[1]);
AssemblyLoadContext.Default.Resolving += (_, name) => File.Exists(Path.Combine(hostRoot, name.Name + ".dll"))
    ? AssemblyLoadContext.Default.LoadFromAssemblyPath(Path.Combine(hostRoot, name.Name + ".dll")) : null;
var baseline = AssemblyLoadContext.Default.LoadFromAssemblyPath(Path.Combine(hostRoot, "Lokad.Utf8Regex.dll"));
var candidate = new AssemblyLoadContext("candidate").LoadFromAssemblyPath(candidatePath);
var catalog = AssemblyLoadContext.Default.LoadFromAssemblyPath(Path.Combine(hostRoot, "Lokad.Utf8Regex.Benchmarks.dll"))
    .GetType("Lokad.Utf8Regex.Benchmarks.ReplicaCountBenchmarkCase")!;
var resolve = catalog.GetMethod("Resolve", BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic)!;
var clock = Stopwatch.StartNew();
var rows = new List<object>();
using var process = Process.GetCurrentProcess();
process.ProcessorAffinity = (nint)4; // CPU2, matching the frozen quiet-VM comparison policy.
foreach (var name in new[] { "miss", "finite-token", "present-miss", "hit", "control" })
{
    var id = name == "control" ? "literal/sherlock-en" : "common/one-node-backtracking";
    var definition = name == "control" ? resolve.Invoke(null, [id])! :
        Activator.CreateInstance(catalog.Assembly.GetType("Lokad.Utf8Regex.Benchmarks.LokadPublicBenchmarkContext")!, [id])!;
    T Get<T>(string property) => (T)definition.GetType().GetProperty(property)!.GetValue(definition)!;
    var input = Get<byte[]>("InputBytes");
    var pattern = Get<string>(name == "control" ? "Utf8Pattern" : "Pattern");
    var options = Get<RegexOptions>("Options") | (name == "ordinary" ? 0 : RegexOptions.Compiled);
    var finite = name == "finite-token";
    if (finite)
    {
        var end = Math.Min(32768, input.Length);
        while (end < input.Length && (input[end] & 0xC0) == 0x80) end--;
        input = input[..end];
    }
    if (name == "present-miss") input = Encoding.UTF8.GetBytes("This regex has the potential to be optimized further.");
    if (name is "hit" or "pcre-hit") input = Encoding.UTF8.GetBytes("ébc.😀tail");
    var pcre = true;
    var text = Encoding.UTF8.GetString(input);
    var oracle = new Regex(pattern, options);
    var expected = oracle.IsMatch(text) ? 1 : 0;
    using var cancellation = new CancellationTokenSource();
    Func<int> Make(Assembly assembly)
    {
        var context = AssemblyLoadContext.GetLoadContext(assembly)!;
        var path = ReferenceEquals(assembly, baseline) ? Path.Combine(hostRoot, "Lokad.Utf8Regex.Pcre2.dll") : Path.GetFullPath(args[2]);
        var adapter = context.Assemblies.FirstOrDefault(a => a.GetName().Name == "Lokad.Utf8Regex.Pcre2") ?? context.LoadFromAssemblyPath(path);
        var profileType = adapter.GetType("Lokad.Utf8Regex.Pcre2.Utf8Pcre2Regex")!;
        if (!finite)
        {
            var owner = profileType.GetConstructor([typeof(string)])!.Invoke([pattern]);
            var isMatch = profileType.GetMethod("IsMatch", [typeof(ReadOnlySpan<byte>)])!.CreateDelegate<SpanPredicate>(owner);
            return () => isMatch(input) ? 1 : 0;
        }
        var policyType = assembly.GetType("Lokad.Utf8Regex.Utf8ResourcePolicy")!;
        var policy = Activator.CreateInstance(policyType)!;
        foreach (var (key, value) in new[] { ("MaxWorkSteps", 10000000UL), ("MaxDepth", 4096UL),
            ("MaxWorkingBytes", 16777216UL), ("MaxResultBytes", 16777216UL) })
        {
            var property = policyType.GetProperty(key)!;
            property.SetValue(policy, Convert.ChangeType(value, Nullable.GetUnderlyingType(property.PropertyType) ?? property.PropertyType));
        }
        var compileOptions = adapter.GetType("Lokad.Utf8Regex.Pcre2.Pcre2CompileOptions")!;
        var settingsType = adapter.GetType("Lokad.Utf8Regex.Pcre2.Utf8Pcre2CompileSettings")!;
        var matchOptions = adapter.GetType("Lokad.Utf8Regex.Pcre2.Pcre2MatchOptions")!;
        var finiteOwner = profileType.GetMethod("Create", [typeof(string), compileOptions, settingsType, policyType, typeof(TimeSpan)])!
            .Invoke(null, [pattern, Activator.CreateInstance(compileOptions), Activator.CreateInstance(settingsType), policy, Regex.InfiniteMatchTimeout]);
        var method = profileType.GetMethod("IsMatch", [typeof(ReadOnlySpan<byte>), typeof(int), matchOptions, typeof(CancellationToken)])!;
        // Bind each ALC's enum outside timing; the timed delegate calls the actual
        // public overload with start 0, None, and the original live token.
        var thunk = new DynamicMethod("PublicPcre2IsMatch", typeof(bool), [typeof(object), typeof(ReadOnlySpan<byte>), typeof(CancellationToken)]);
        var il = thunk.GetILGenerator();
        il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, profileType);
        il.Emit(OpCodes.Ldarg_1); il.Emit(OpCodes.Ldc_I4_0); il.Emit(OpCodes.Ldc_I4_0);
        il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Callvirt, method); il.Emit(OpCodes.Ret);
        var call = thunk.CreateDelegate<TokenPredicate>(finiteOwner);
        return () => call(input, cancellation.Token) ? 1 : 0;
    }
    Func<int>[] actions = [Make(baseline), Make(candidate), () => oracle.IsMatch(text) ? 1 : 0,
        () => oracle.IsMatch(Encoding.UTF8.GetString(input)) ? 1 : 0];
    var iterations = new int[4];
    var warmupCalls = new int[4];
    var warmupMs = new double[4];
    for (var lane = 0; lane < actions.Length; lane++)
    {
        if (actions[lane]() != expected) throw new Exception("Preflight mismatch");
        var warmup = Stopwatch.StartNew();
        do { Loop(1, actions[lane]); warmupCalls[lane]++; } while (warmup.ElapsedMilliseconds < 250);
        warmupMs[lane] = warmup.Elapsed.TotalMilliseconds;
        var count = 1;
        var probe = Loop(count, actions[lane]);
        while (probe.Milliseconds < 20 && count < 1000000)
        { count = Math.Min(count * 2, 1000000); probe = Loop(count, actions[lane]); }
        iterations[lane] = (int)Math.Clamp(Math.Ceiling(count * 50 / Math.Max(probe.Milliseconds, 0.000001)), 1, 1000000);
    }
    var times = Enumerable.Range(0, 4).Select(_ => new double[3]).ToArray();
    var allocations = Enumerable.Range(0, 4).Select(_ => new double[3]).ToArray();
    for (var sample = 0; sample < 3; sample++)
    for (var position = 0; position < 4; position++)
    {
        var lane = sample % 2 == 0 ? position : 3 - position;
        GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
        var result = Loop(iterations[lane], actions[lane]);
        if (result.Last != expected) throw new Exception("Timed result mismatch");
        times[lane][sample] = result.Milliseconds * 1000 / iterations[lane];
        allocations[lane][sample] = (double)result.Bytes / iterations[lane];
    }
    rows.Add(new { Name = name, Id = id, Options = pcre ? "Pcre2 defaults; .NET comparator Compiled" : options.ToString(), Profile = pcre ? "Pcre2" : "Core", FiniteAndLiveToken = finite,
        InputBytes = input.Length, InputSha256 = Convert.ToHexString(SHA256.HashData(input)), Expected = expected,
        Iterations = iterations, WarmupCalls = warmupCalls, WarmupMilliseconds = warmupMs,
        Microseconds = times, AllocatedBytes = allocations });
    Console.Error.WriteLine($"Completed {name}: {string.Join(" / ", times.Select(t => t.Order().ElementAt(1)))} us/op");
}
Console.WriteLine(JsonSerializer.Serialize(new { Runtime = RuntimeInformation.FrameworkDescription,
    CpuAffinityMask = "4", ElapsedSeconds = clock.Elapsed.TotalSeconds,
    LaneNames = new[] { "Baseline", "Candidate", "PredecodedRegex", "DecodeThenRegex" },
    Baseline = Identity(baseline), Candidate = Identity(candidate),
    BaselinePcre2 = Identity(AssemblyLoadContext.GetLoadContext(baseline)!.Assemblies.Single(a => a.GetName().Name == "Lokad.Utf8Regex.Pcre2")),
    CandidatePcre2 = Identity(AssemblyLoadContext.GetLoadContext(candidate)!.Assemblies.Single(a => a.GetName().Name == "Lokad.Utf8Regex.Pcre2")), Rows = rows }, new JsonSerializerOptions { WriteIndented = true }));
static object Identity(Assembly a) => new { Version = a.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion,
    Sha256 = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(a.Location))) };
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes, int Last) Loop(int count, Func<int> action)
{
    var value = 0; var before = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) value = action();
    var elapsed = Stopwatch.GetElapsedTime(start);
    return (elapsed.TotalMilliseconds, GC.GetAllocatedBytesForCurrentThread() - before, value);
}
delegate bool SpanPredicate(ReadOnlySpan<byte> input);
delegate bool TokenPredicate(ReadOnlySpan<byte> input, CancellationToken token);
