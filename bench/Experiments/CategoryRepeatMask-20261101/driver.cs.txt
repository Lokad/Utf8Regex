[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Measure(int count, Func<int> action, int expected) {
    var values = new int[count]; var before = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) values[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(start); var bytes = GC.GetAllocatedBytesForCurrentThread() - before;
    foreach (var value in values) if (value != expected) throw new Exception("Timed result mismatch");
    GC.KeepAlive(values); return (elapsed.TotalMilliseconds, bytes);
}
