using System.Text;
using System.Text.Json;
using System.Text.RegularExpressions;
using Lokad.Utf8Regex;
var input=Encoding.UTF8.GetBytes("word abcdefghij 0123456789_ xxx");
foreach(var pattern in new[]{@"\b\w{10,}\b",@"\w{10,}",@"\w{10,}?",@"\b\w{10,}?\b",@"(\b\w{10,}\b)",@"[\w-]{10,}"})
foreach(var options in new[]{RegexOptions.None,RegexOptions.Compiled,RegexOptions.NonBacktracking,RegexOptions.RightToLeft,RegexOptions.ECMAScript})
{
    var regex=new Utf8Regex(pattern,options);var result=regex.CollectCountDiagnostics(input);
    Console.WriteLine(JsonSerializer.Serialize(new{Pattern=pattern,Options=options.ToString(),Family=regex.Inspection.DebugFallbackDirectFamilyKind,result.ExecutionRoute,result.Result}));
}
