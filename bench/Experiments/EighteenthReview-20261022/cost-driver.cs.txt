const string identifierPattern = @"\b(?:record|struct|class)\s+[A-Z][A-Za-z0-9_]+";
const string namedPattern = @"(?<x>a)(?=(?=(?=(?<y>b))))(?<z>b)\k<x>";
var text = string.Concat(Enumerable.Repeat("record Bravo struct Delta class Echo public async KEYWORD needle é😀!\n",8));
var bytes = Encoding.UTF8.GetBytes(text);
var namedBytes = Encoding.UTF8.GetBytes(string.Concat(Enumerable.Repeat("aba ",32)));
var ordinary = new Utf8Regex(identifierPattern, RegexOptions.CultureInvariant);
var compiled = new Utf8Regex(identifierPattern, RegexOptions.CultureInvariant | RegexOptions.Compiled);
var finite = new Utf8Regex(identifierPattern, RegexOptions.CultureInvariant | RegexOptions.Compiled, Timeout.InfiniteTimeSpan,
    new Utf8ResourcePolicy { MaxWorkSteps=1_000_000, MaxDepth=2048, MaxWorkingBytes=16*1024*1024, MaxResultBytes=16*1024*1024 });
var window = new Utf8Regex(@"\bpublic\s+async\b",RegexOptions.CultureInvariant | RegexOptions.Compiled);
var folded = new Utf8Regex(@"\bkeyword\b",RegexOptions.CultureInvariant | RegexOptions.IgnoreCase);
var literal = new Utf8Regex("needle",RegexOptions.CultureInvariant | RegexOptions.Compiled);
var named = new Utf8Pcre2Regex(namedPattern);
var character = new Utf8Pcre2Regex("[é]");
var oracle = new Regex(identifierPattern,RegexOptions.CultureInvariant);
var strict = new UTF8Encoding(false,true);
var lanes = new[] {
    new Lane("construction/pcre2-named-nested-assertions", () => Construct(namedPattern), 4, 16),
    new Lane("public/pcre2-Analyze-character", () => character.Analyze().MinRequiredLengthInBytes, args.Contains("--fixed") ? 1 : 0, 100_000),
    new Lane("public/pcre2-Count-named-assertions", () => named.Count(namedBytes), 32, 32),
    new Lane("public/core-identifier-ordinary-Count", () => ordinary.Count(bytes), 24, 256),
    new Lane("public/core-identifier-compiled-Count", () => compiled.Count(bytes), 24, 256),
    new Lane("public/core-identifier-finite-four-Count", () => finite.Count(bytes), 24, 4),
    new Lane("public/core-window-compiled-Count", () => window.Count(bytes), 8, 256),
    new Lane("public/core-window-compiled-IsMatch", () => window.IsMatch(bytes) ? 1 : 0, 1, 100_000),
    new Lane("public/core-folded-boundary-Count", () => folded.Count(bytes), 8, 256),
    new Lane("public/core-literal-compiled-Enumerate", () => Enumerate(literal,bytes), 8, 1024),
    new Lane("control/unchanged-literal-Count", () => literal.Count(bytes), 8, 1024),
    new Lane("control/predecoded-Regex-Count", () => oracle.Count(text), 24, 1024),
    new Lane("control/decode-then-Regex-Count", () => oracle.Count(strict.GetString(bytes)), 24, 1024)
};
if (args.Contains("--preflight")) {
    Console.WriteLine(JsonSerializer.Serialize(lanes.Select(l => new {l.Id,l.Expected,Sample=Run(l,1)}),new JsonSerializerOptions {WriteIndented=true}));

}
foreach (var lane in lanes) {
    Run(lane,1);
    var warm = Stopwatch.StartNew();
    do { Run(lane,lane.Minimum); } while(warm.ElapsedMilliseconds<750);
    lane.Iterations=lane.Minimum;
    for(var attempt=0;attempt<4;attempt++) {
        var sample=Run(lane,lane.Iterations);
        if(sample.Milliseconds>=35)break;
        lane.Iterations=(int)Math.Clamp(Math.Ceiling(lane.Iterations*45/Math.Max(.01,sample.Milliseconds)),lane.Minimum,2_000_000);
    }
}
for(var sample=0;sample<5;sample++)
    foreach(var lane in sample%2==0 ? lanes : lanes.Reverse()) lane.Samples.Add(Run(lane,lane.Iterations));
Console.WriteLine(JsonSerializer.Serialize(new {

static int Construct(string pattern) {var regex=new Utf8Pcre2Regex(pattern);var names=regex.NameEntryCount;GC.KeepAlive(regex);return names+1;}
static int Enumerate(Utf8Regex regex,byte[] input) {var cursor=regex.EnumerateMatches(input);var count=0;while(cursor.MoveNext())count++;return count;}
static Sample Run(Lane lane,int iterations) {
    var allocated=GC.GetAllocatedBytesForCurrentThread();var start=Stopwatch.GetTimestamp();long checksum=0;
    for(var iteration=0;iteration<iterations;iteration++)checksum+=lane.Body();
    var elapsed=Stopwatch.GetElapsedTime(start).TotalMilliseconds;allocated=GC.GetAllocatedBytesForCurrentThread()-allocated;
    if(checksum!=(long)lane.Expected*iterations)throw new Exception("Result mismatch: "+lane.Id);
    return new Sample(elapsed,elapsed*1_000_000/iterations,(double)allocated/iterations,checksum);
}
sealed class Lane(string id,Func<int> body,int expected,int minimum) {
    internal string Id{get;}=id;internal Func<int> Body{get;}=body;internal int Expected{get;}=expected;
    internal int Minimum{get;}=minimum;internal int Iterations{get;set;}internal List<Sample> Samples{get;}=[];
}
sealed record Sample(double Milliseconds,double NanosecondsPerOperation,double AllocatedBytesPerOperation,long Checksum);
