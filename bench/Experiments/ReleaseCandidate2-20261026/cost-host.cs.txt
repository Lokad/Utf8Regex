using System.Diagnostics;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Runtime.Loader;
using System.Security.Cryptography;
using System.Text.Json;
using Lokad.Utf8Regex.Benchmarks;

var started = Stopwatch.StartNew();
var root = Path.GetFullPath(args[0]);
var sampleCount = args.Length > 1 ? int.Parse(args[1]) : 9;
var confirmation = sampleCount > 9;
var warmupMilliseconds = 2000;
using var processorScope = BenchmarkProcessorScope.EnterSingleHighestEfficiencyProcessor();
var driver = Path.Combine(root, "cost-driver/bin/Release/net10.0/cost-driver.dll");
var contexts = new[] { new VersionContext(Path.Combine(root, "before")), new VersionContext(Path.Combine(root, "after")) };
var types = contexts.Select(context => context.LoadFromAssemblyPath(driver).GetType("Lanes")!).ToArray();
var versions = types.Select(type => (string)type.GetMethod("Producer")!.Invoke(null, null)!).ToArray();
var lanes = types.Select(type => ((string Name, Func<long> Run, long Expected)[])type.GetMethod("Create")!.Invoke(null, null)!).ToArray();
foreach (var versionLanes in lanes)
    foreach (var lane in versionLanes)
        if (lane.Run() != lane.Expected) throw new Exception("Public preflight mismatch: " + lane.Name);
if (args.Contains("--preflight")) {
    Console.WriteLine(JsonSerializer.Serialize(new { Versions=versions,
        Lanes=lanes[0].Select(lane=>new {lane.Name,lane.Expected}), Preflight="All selected public outputs agree with independent expected checksums" },
        new JsonSerializerOptions {WriteIndented=true}));
    return;
}
if (args.Length > 2) {
    var selected = args[2].Split(',').ToHashSet(StringComparer.Ordinal);
    lanes = lanes.Select(rows => rows.Where(row => selected.Contains(row.Name)).ToArray()).ToArray();
}
var rows = new List<object>();
long sink = 0;
for (var laneIndex = 0; laneIndex < lanes[0].Length; laneIndex++)
{
    if (started.Elapsed.TotalSeconds > 180) throw new Exception("Bounded comparison exceeded 180 seconds");
    var pair = new[] { lanes[0][laneIndex], lanes[1][laneIndex] };
    if (pair[0].Name != pair[1].Name || pair[0].Expected != pair[1].Expected) throw new Exception("Lane mismatch");
    var counts = new int[2];
    foreach (var version in confirmation ? new[] { 1, 0 } : new[] { 0, 1 })
    {
        if (pair[version].Run() != pair[version].Expected) throw new Exception("Public result mismatch: " + pair[version].Name);
        var warm = Stopwatch.StartNew();
        long warmCalls = 0; do { for (var n = 0; n < 100; n++) { sink += pair[version].Run(); warmCalls++; } } while (warm.ElapsedMilliseconds < warmupMilliseconds || warmCalls < (pair[version].Name.Contains("short-") ? 5_000_000 : 10_000));
        var count = 1;
        while (true)
        {
            var measured = Measure(pair[version].Run, count);
            if (measured.Nanoseconds * count >= 40_000_000) break;
            count = checked((int)Math.Min(16_000_000, Math.Max(count + 1, count * Math.Min(8, 45_000_000 / Math.Max(1, measured.Nanoseconds * count)))));
        }
        counts[version] = count;
    }
    var samples = new List<object>();
    var timings = new[] { new List<double>(), new List<double>() };
    var byteCosts = new[] { new List<double>(), new List<double>() };
    for (var sample = 0; sample < sampleCount; sample++)
    {
        foreach (var version in (sample + (confirmation ? 1 : 0)) % 2 == 0 ? new[] { 0, 1 } : new[] { 1, 0 })
        {
            var measured = Measure(pair[version].Run, counts[version]);
            if (measured.Checksum != pair[version].Expected * counts[version]) throw new Exception("Timed checksum mismatch");
            timings[version].Add(measured.Nanoseconds);
            byteCosts[version].Add(measured.Bytes);
            samples.Add(new { Sample = sample, Version = version, Count = counts[version], measured.Nanoseconds, measured.Bytes, measured.Checksum });
        }
    }
    var before = Median(timings[0]);
    var after = Median(timings[1]);
    rows.Add(new { Name = pair[0].Name, BeforeNs = before, AfterNs = after, ChangePercent = (after / before - 1) * 100,
        BeforeBytes = Median(byteCosts[0]), AfterBytes = Median(byteCosts[1]), BeforeSpread = timings[0].Max() / timings[0].Min(), AfterSpread = timings[1].Max() / timings[1].Min(), Samples = samples });
}
Console.WriteLine(JsonSerializer.Serialize(new {
    Scope = "One bounded same-process alternating before/after public comparison; final-package versus fixed correctness-equivalent 844dcbad cumulative comparison.",
    Runtime = RuntimeInformation.FrameworkDescription, OS = RuntimeInformation.OSDescription,
    Versions = versions, CoreHashes = new[] { "before", "after" }.Select(path => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(Path.Combine(root, path, "Lokad.Utf8Regex.dll"))))).ToArray(),
    ProcessorPolicy = processorScope.Policy, AffinityMask = processorScope.AffinityMask, Processor = processorScope.Description,
    Construction = types.Select(type => type.GetMethod("Construction")!.Invoke(null,null)).ToArray(),
    SampleCount = sampleCount, WarmupMillisecondsPerVersionPerLane = warmupMilliseconds, MinimumCalibrationMilliseconds = 40,
    WallSeconds = started.Elapsed.TotalSeconds, Rows = rows, Sink = sink
}, new JsonSerializerOptions { WriteIndented = true }));

static (double Nanoseconds, double Bytes, long Checksum) Measure(Func<long> run, int count)
{
    var allocated = GC.GetAllocatedBytesForCurrentThread();
    var start = Stopwatch.GetTimestamp();
    long sum = 0;
    for (var n = 0; n < count; n++) sum += run();
    var elapsed = Stopwatch.GetTimestamp() - start;
    return (elapsed * (1e9 / Stopwatch.Frequency) / count, (GC.GetAllocatedBytesForCurrentThread() - allocated) / (double)count, sum);
}
static double Median(List<double> samples) => samples.Order().ElementAt(samples.Count / 2);
sealed class VersionContext(string directory) : AssemblyLoadContext
{
    protected override Assembly? Load(AssemblyName name) => name.Name is "Lokad.Utf8Regex" or "Lokad.Utf8Regex.PythonRe" or "Lokad.Utf8Regex.Pcre2"
        ? LoadFromAssemblyPath(Path.Combine(directory, name.Name + ".dll")) : null;
}
