static Sample Run(Lane lane,int iterations) {
 var a=GC.GetAllocatedBytesForCurrentThread();var start=Stopwatch.GetTimestamp();long sum=0;
 for(int i=0;i<iterations;i++)sum+=lane.Action();

 if(sum!=(long)iterations*lane.Expected)throw new Exception("Wrong result: "+lane.Id);
 return new(elapsed*1e9/Stopwatch.Frequency/iterations,(double)a/iterations,elapsed*1000.0/Stopwatch.Frequency,sum);
}
