    !after.Python.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion.Contains(args[2], StringComparison.Ordinal))

    throw new Exception("Unexpected corpus bytes");
string Repeat(string value, int count) => string.Concat(Enumerable.Repeat(value, count));

    new Case("dense-ascii", Mail, Repeat("one.two+-@domain.example ", 128), 128),
    new Case("dense-unicode", Mail, Repeat("é²𐐀._+-@dömain.exämple ", 96), 96),

var elapsed = Stopwatch.StartNew(); var preflightOnly = args.Length > 4 && args[4] == "--preflight";
var confirmation = args.Length > 4 && args[4] == "--confirm";
var rows = new List<object>();
foreach (var item in cases)
{
    if (confirmation && item.Name is not ("mail-corpus" or "finite-unicode" or "complement-control" or "class-program-control" or "literal-control" or "core-boundary-control")) continue;
    var input = Encoding.UTF8.GetBytes(item.Text);
    using var cancellation = new CancellationTokenSource();
    Func<int> Make((Assembly Core, Assembly Python) assemblies)
    {
        var type = item.Core ? assemblies.Core.GetType("Lokad.Utf8Regex.Utf8Regex")! : assemblies.Python.GetType("Lokad.Utf8Regex.PythonRe.Utf8PythonRegex")!;
        object owner;
        if (item.Finite)
        {
            var optionsType = assemblies.Python.GetType("Lokad.Utf8Regex.PythonRe.PythonReCompileOptions")!;

                ("MaxWorkingBytes", 67108864UL), ("MaxResultBytes", 67108864UL) })
            {
                var property = policyType.GetProperty(key)!;
                property.SetValue(policy, Convert.ChangeType(value, Nullable.GetUnderlyingType(property.PropertyType) ?? property.PropertyType));
            }

                .Invoke([item.Pattern, Enum.ToObject(optionsType, 0), Timeout.InfiniteTimeSpan, policy]);
            var call = type.GetMethod("Count", [typeof(ReadOnlySpan<byte>), typeof(int), typeof(CancellationToken)])!.CreateDelegate<TokenCount>(owner);
            return () => call(input, 0, cancellation.Token);
        }
        owner = type.GetConstructor([typeof(string)])!.Invoke([item.Pattern]);
        var count = type.GetMethod("Count", [typeof(ReadOnlySpan<byte>)])!.CreateDelegate<SpanCount>(owner);
        return () => count(input);
    }
    var baseline = Make(before); var candidate = Make(after);
    var oracle = item.DotNetComparable ? new Regex(item.Pattern, RegexOptions.CultureInvariant) : null;
    Func<int>[] actions = oracle is null ? [baseline, candidate]

    foreach (var action in actions) if (action() != item.Expected) throw new Exception($"Preflight mismatch: {item.Name}");
    if (preflightOnly) { Console.Error.WriteLine($"Preflight {item.Name}: {item.Expected}, bytes={input.Length}"); continue; }

    for (var lane = 0; lane < lanes; lane++)

        warmupMilliseconds[lane] = warmup.Elapsed.TotalMilliseconds;

        while (probe.Milliseconds < 20 && count < 1000000) { count *= 2; probe = Loop(count, actions[lane]); }
        iterations[lane] = (int)Math.Clamp(Math.Ceiling(count * 50 / Math.Max(probe.Milliseconds, 0.000001)), 1, 1000000);
    }

    var allocations = Enumerable.Range(0, lanes).Select(_ => new double[3]).ToArray();
    var durations = Enumerable.Range(0, lanes).Select(_ => new double[3]).ToArray();

    for (var position = 0; position < lanes; position++)
    {
        var lane = sample % 2 == 0 ? position : lanes - 1 - position;
        GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
        var result = Loop(iterations[lane], actions[lane]); if (result.Last != item.Expected) throw new Exception("Timed mismatch");
        times[lane][sample] = result.Milliseconds * 1000 / iterations[lane];
        allocations[lane][sample] = (double)result.Bytes / iterations[lane]; durations[lane][sample] = result.Milliseconds;
    }

        Iterations = iterations, WarmupCalls = warmupCalls, WarmupMilliseconds = warmupMilliseconds,

}

static (Assembly Core, Assembly Python) Load(string root, string name)
{
    var context = new AssemblyLoadContext(name);
    context.Resolving += (ctx, n) => File.Exists(Path.Combine(root, n.Name + ".dll")) ? ctx.LoadFromAssemblyPath(Path.Combine(root, n.Name + ".dll")) : null;
    return (context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.dll")), context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.PythonRe.dll")));
}
static string Hash(Assembly a) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(a.Location)));
static object Identity(Assembly a) => new { Version = a.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion, Sha256 = Hash(a) };
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes, int Last) Loop(int count, Func<int> action)
{
    var value = 0; var bytes = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) value = action();
    return (Stopwatch.GetElapsedTime(start).TotalMilliseconds, GC.GetAllocatedBytesForCurrentThread() - bytes, value);
}
