static int Enumerate(Utf8Pcre2Regex regex,byte[] input)
{
    var cursor=regex.EnumerateMatches(input);var total=0;
    while(cursor.MoveNext())
    {
        var current=cursor.Current;
        if(!current.Success || !current.IsUtf8SliceWellFormed || !current.HasUtf16Projection) throw new InvalidOperationException("Invalid public range.");
        total+=current.EndOffsetInBytes-current.StartOffsetInBytes;
    }
    return total;
}
static int CopyNames(Utf8Pcre2Regex regex)
{
    Span<int> destination=stackalloc int[1];
    var count=regex.CopyNumbersForName("n",destination,out var more);
    if(!more || destination[0]!=1) throw new InvalidOperationException("Invalid named slots.");
    return count;
}
static int CopyContext(scoped in Utf8Pcre2MatchContext context)
{
    Span<int> destination=stackalloc int[1];
    var count=context.CopyNumbersForName("n",destination,out var more);
    if(!more || destination[0]!=1) throw new InvalidOperationException("Invalid named slots.");
    return count;
}
static Sample Run(Lane lane,int iterations,scoped in Utf8Pcre2MatchContext context)
{
    var allocated=GC.GetAllocatedBytesForCurrentThread();var start=Stopwatch.GetTimestamp();long checksum=0;
    if(lane.Action is { } action)
        for(var i=0;i<iterations;i++) checksum+=action();
    else
        for(var i=0;i<iterations;i++) checksum+=CopyContext(in context);
    var ticks=Stopwatch.GetTimestamp()-start;allocated=GC.GetAllocatedBytesForCurrentThread()-allocated;
    if(checksum!=(long)iterations*lane.Expected)throw new InvalidOperationException("Result changed: "+lane.Id);
    return new(ticks*1e9/Stopwatch.Frequency/iterations,(double)allocated/iterations,ticks*1000.0/Stopwatch.Frequency,checksum);
}
sealed class Lane(string id,Func<int>? action,int expected,int inputBytes,int minimumCalls)
{
    internal string Id=id;internal Func<int>? Action=action;internal int Expected=expected;internal int InputBytes=inputBytes;
    internal int Iterations;internal int MinimumCalls=minimumCalls;internal List<Sample> Samples=[];
}
sealed record Sample(double NanosecondsPerOperation,double AllocatedBytesPerOperation,double WallMilliseconds,long Checksum);
