static Func<object, byte[], string> ProjectionBridge(Assembly assembly) {
    var regex = assembly.GetType("Lokad.Utf8Regex.Utf8Regex")!;
    var cursor = assembly.GetType("Lokad.Utf8Regex.Internal.Execution.Utf8OperationMatchCursor")!;
    var validation = assembly.GetType("Lokad.Utf8Regex.Internal.Input.Utf8Validation")!;
    var factory = assembly.GetType("Lokad.Utf8Regex.Internal.Execution.Utf8CompiledOperationCursorFactory")!;
    var prepared = regex.GetProperty("_preparedRegex", BindingFlags.Instance | BindingFlags.NonPublic)!.GetMethod!;
    var validate = validation.GetMethod("Validate", BindingFlags.Public | BindingFlags.Static, [typeof(ReadOnlySpan<byte>)])!;
    var deadline = regex.GetMethod("CreateExecutionBudget", BindingFlags.NonPublic | BindingFlags.Instance, Type.EmptyTypes)!;
    var create = factory.GetMethod("CreateMatchCursor", BindingFlags.Public | BindingFlags.Static,
        [prepared.ReturnType, typeof(ReadOnlySpan<byte>), validate.ReturnType, deadline.ReturnType])!;
    var field = cursor.GetField("_projectionPlan", BindingFlags.Instance | BindingFlags.NonPublic)!;
    var kind = field.FieldType.GetProperty("Kind")!.GetMethod!;
    var bridge = new DynamicMethod("ProjectionCoverage", typeof(string), [typeof(object), typeof(byte[])], typeof(Program).Module, true);
    var il = bridge.GetILGenerator(); var local = il.DeclareLocal(cursor); var metadata = il.DeclareLocal(validate.ReturnType);
    ReadSpan(il, 1); il.Emit(OpCodes.Call, validate); il.Emit(OpCodes.Stloc, metadata);
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, regex); il.Emit(OpCodes.Call, prepared); ReadSpan(il, 1); il.Emit(OpCodes.Ldloc, metadata);
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, regex); il.Emit(OpCodes.Call, deadline); il.Emit(OpCodes.Call, create); il.Emit(OpCodes.Stloc, local);
    il.Emit(OpCodes.Ldloca, local); il.Emit(OpCodes.Ldflda, field); il.Emit(OpCodes.Call, kind); il.Emit(OpCodes.Box, kind.ReturnType);
    il.Emit(OpCodes.Callvirt, typeof(object).GetMethod("ToString")!); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<Func<object, byte[], string>>();
}

static Func<object, byte[], int> CountBridge(Type type) {
    var bridge = new DynamicMethod("PublicCount", typeof(int), [typeof(object), typeof(byte[])], typeof(Program).Module, true); var il = bridge.GetILGenerator();

}
static Func<object, byte[], object, CancellationToken, object> ReplaceBridge(Type type, bool native, bool text) {
    var method = type.GetMethod(text ? "ReplaceToString" : "Replace", [typeof(ReadOnlySpan<byte>), native ? typeof(ReadOnlySpan<byte>) : typeof(string), typeof(CancellationToken)])!;
    var bridge = new DynamicMethod("PublicReplace", typeof(object), [typeof(object), typeof(byte[]), typeof(object), typeof(CancellationToken)], typeof(Program).Module, true); var il = bridge.GetILGenerator();
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); ReadSpan(il, 1); il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Castclass, native ? typeof(byte[]) : typeof(string));
    if (native) il.Emit(OpCodes.Call, typeof(ReadOnlySpan<byte>).GetMethod("op_Implicit", [typeof(byte[])])!);
    il.Emit(OpCodes.Ldarg_3); il.Emit(OpCodes.Call, method); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<Func<object, byte[], object, CancellationToken, object>>();
}
static Func<object, byte[], object, byte[], int> DestinationBridge(Type type, bool native) {
    var method = type.GetMethod("TryReplace", [typeof(ReadOnlySpan<byte>), native ? typeof(ReadOnlySpan<byte>) : typeof(string), typeof(Span<byte>), typeof(int).MakeByRefType()])!;
    var bridge = new DynamicMethod("PublicTryReplace", typeof(int), [typeof(object), typeof(byte[]), typeof(object), typeof(byte[])], typeof(Program).Module, true); var il = bridge.GetILGenerator();
    var written = il.DeclareLocal(typeof(int)); var done = il.DefineLabel();
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); ReadSpan(il, 1); il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Castclass, native ? typeof(byte[]) : typeof(string));
    if (native) il.Emit(OpCodes.Call, typeof(ReadOnlySpan<byte>).GetMethod("op_Implicit", [typeof(byte[])])!);
    il.Emit(OpCodes.Ldarg_3); il.Emit(OpCodes.Call, typeof(Span<byte>).GetMethod("op_Implicit", [typeof(byte[])])!); il.Emit(OpCodes.Ldloca, written);
    il.Emit(OpCodes.Call, method); il.Emit(OpCodes.Brfalse, done); il.Emit(OpCodes.Ldc_I4_M1); il.Emit(OpCodes.Ret); il.MarkLabel(done); il.Emit(OpCodes.Ldloc, written); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<Func<object, byte[], object, byte[], int>>();
}
static Func<object, byte[], object, CancellationToken, bool> SelectorBridge(Type type, Type plan) {
    var bridge = new DynamicMethod("RouteProof", typeof(bool), [typeof(object), typeof(byte[]), typeof(object), typeof(CancellationToken)], typeof(Program).Module, true); var il = bridge.GetILGenerator();
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, type); ReadSpan(il, 1); il.Emit(OpCodes.Ldarg_2); il.Emit(OpCodes.Unbox_Any, plan); il.Emit(OpCodes.Ldarg_3);
    il.Emit(OpCodes.Call, type.GetMethod("GetOwnedReplacementProgram", BindingFlags.NonPublic | BindingFlags.Instance)!); il.Emit(OpCodes.Ldnull); il.Emit(OpCodes.Cgt_Un); il.Emit(OpCodes.Ret);
    return bridge.CreateDelegate<Func<object, byte[], object, CancellationToken, bool>>();
}

[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Loop(int count, Func<Output> action, Action<Output> check) {
    var output = new Output[count]; var bytes = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp(); for (var i = 0; i < count; i++) output[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(start).TotalMilliseconds; bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
    foreach (var value in output) check(value); GC.KeepAlive(output); return (elapsed, bytes);
}
