static (Assembly Core, Assembly Python) Load(string root, string name) {
    var context = new AssemblyLoadContext(name); context.Resolving += (c, n) => File.Exists(Path.Combine(root, n.Name + ".dll")) ? c.LoadFromAssemblyPath(Path.Combine(root, n.Name + ".dll")) : null;
    return (context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.dll")), context.LoadFromAssemblyPath(Path.Combine(root, "Lokad.Utf8Regex.PythonRe.dll")));
}
static string Version(Assembly a) => a.GetCustomAttribute<AssemblyInformationalVersionAttribute>()!.InformationalVersion;
static string Hash(Assembly a) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(a.Location)));
static object Identity(Assembly a) => new { Version = Version(a), Sha256 = Hash(a) };
[MethodImpl(MethodImplOptions.NoInlining)]
static (double Milliseconds, long Bytes) Measure(int count, Func<long> action, long expected) {
    var values = new long[count]; var before = GC.GetAllocatedBytesForCurrentThread(); var start = Stopwatch.GetTimestamp();
    for (var i = 0; i < count; i++) values[i] = action();
    var elapsed = Stopwatch.GetElapsedTime(start); var bytes = GC.GetAllocatedBytesForCurrentThread() - before;
    foreach (var value in values) if (value != expected) throw new Exception("Timed result mismatch");
    GC.KeepAlive(values); return (elapsed.TotalMilliseconds, bytes);
}
