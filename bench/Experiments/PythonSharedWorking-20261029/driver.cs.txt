static PublicLoop Bridge(Type ownerType, bool useMoveToken, Type? allowanceType)
{
    var factory = ownerType.GetMethod("CreateMatchCursor", allowanceType is null
        ? [typeof(ReadOnlyMemory<byte>), typeof(int), typeof(CancellationToken)]

    var detailedType = currentGetter.ReturnType; var groupsGetter = detailedType.GetProperty("Groups")!.GetMethod!;
    var arrayType = groupsGetter.ReturnType; var groupType = arrayType.GetElementType()!;
    var method = new DynamicMethod("PublicCursorLoop", typeof(int), [typeof(object), typeof(byte[]), typeof(CancellationToken), typeof(CancellationToken), typeof(object)], typeof(Program).Module, true);

    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Castclass, ownerType); il.Emit(OpCodes.Ldarg_1);
    il.Emit(OpCodes.Newobj, typeof(ReadOnlyMemory<byte>).GetConstructor([typeof(byte[])])!); il.Emit(OpCodes.Ldc_I4_0); il.Emit(OpCodes.Ldarg_2);
    if (allowanceType is not null) { il.Emit(OpCodes.Ldarg_S, 4); il.Emit(OpCodes.Castclass, allowanceType); }
    il.Emit(OpCodes.Callvirt, factory); il.Emit(OpCodes.Stloc, cursor);

    il.Emit(OpCodes.Br, move); il.MarkLabel(next);
    il.Emit(OpCodes.Ldloc, cursor); il.Emit(OpCodes.Callvirt, currentGetter); il.Emit(OpCodes.Stloc, current);
    il.Emit(OpCodes.Ldloca, current); il.Emit(OpCodes.Call, groupsGetter); il.Emit(OpCodes.Stloc, groups);
    il.Emit(OpCodes.Ldc_I4_0); il.Emit(OpCodes.Stloc, index); il.Emit(OpCodes.Br, groupLoop);

    il.MarkLabel(skip); il.Emit(OpCodes.Ldloc, index); il.Emit(OpCodes.Ldc_I4_1); il.Emit(OpCodes.Add); il.Emit(OpCodes.Stloc, index);
    il.MarkLabel(groupLoop); il.Emit(OpCodes.Ldloc, index); il.Emit(OpCodes.Ldloc, groups); il.Emit(OpCodes.Ldlen); il.Emit(OpCodes.Conv_I4); il.Emit(OpCodes.Blt, read);

static object MakeAllowance(Type interfaceType, DriverAllowance host)
{
    var assembly = AssemblyBuilder.DefineDynamicAssembly(new AssemblyName("SharedAllowance" + Guid.NewGuid().ToString("N")), AssemblyBuilderAccess.Run);
    var type = assembly.DefineDynamicModule("SharedAllowance").DefineType("Provider", TypeAttributes.Public | TypeAttributes.Sealed, typeof(object), [interfaceType]);
    var field = type.DefineField("_host", typeof(DriverAllowance), FieldAttributes.Private | FieldAttributes.InitOnly);
    var ctor = type.DefineConstructor(MethodAttributes.Public, CallingConventions.Standard, [typeof(DriverAllowance)]);
    var il = ctor.GetILGenerator(); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Call, typeof(object).GetConstructor(Type.EmptyTypes)!);
    il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Ldarg_1); il.Emit(OpCodes.Stfld, field); il.Emit(OpCodes.Ret);
    foreach (var declaration in interfaceType.GetMethods())
    {
        var method = type.DefineMethod(declaration.Name, MethodAttributes.Public | MethodAttributes.Virtual | MethodAttributes.Final | MethodAttributes.NewSlot,
            declaration.ReturnType, [typeof(ulong)]);
        il = method.GetILGenerator(); il.Emit(OpCodes.Ldarg_0); il.Emit(OpCodes.Ldfld, field); il.Emit(OpCodes.Ldarg_1);
        il.Emit(OpCodes.Callvirt, typeof(DriverAllowance).GetMethod(declaration.Name)!); il.Emit(OpCodes.Ret);
        type.DefineMethodOverride(method, declaration);
    }
    return Activator.CreateInstance(type.CreateType()!, host)!;
}

static (double Milliseconds, long Bytes, int Last) Loop(int count, Func<int> action)
{

public sealed class DriverAllowance(long maximum)
{
    private long _used;
    public long Used => Volatile.Read(ref _used);
    public bool TryReserve(ulong bytes)
    {
        if (bytes > long.MaxValue) return false;
        var delta = (long)bytes;
        while (true)
        {
            var current = Volatile.Read(ref _used);
            if (delta > maximum - current) return false;
            if (Interlocked.CompareExchange(ref _used, current + delta, current) == current) return true;
        }
    }
    public void Release(ulong bytes)
    {
        if (Interlocked.Add(ref _used, -checked((long)bytes)) < 0) throw new Exception("Shared allowance underflow");
    }
}
